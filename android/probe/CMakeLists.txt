cmake_minimum_required(VERSION 3.25)
project(pgr4_vulkan_probe LANGUAGES C)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use the Android NDK toolchain with ANDROID_ABI=arm64-v8a")
endif()
add_executable(pgr4-vulkan-probe main.c)
target_compile_features(pgr4-vulkan-probe PRIVATE c_std_11)
target_compile_options(pgr4-vulkan-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(pgr4-vulkan-probe PRIVATE vulkan)
